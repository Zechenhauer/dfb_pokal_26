async function fetchLiveScores() { try { const response = await fetch('https://site.api.espn.com/apis/site/v2/sports/soccer/ger.dfb_pokal/scoreboard'); const data = await response.json(); let hasLiveGames = false; if (data && data.events) { data.events.forEach(event => { const comp = event.competitions[0]; const homeTeam = comp.competitors.find(c => c.homeAway === 'home'); const awayTeam = comp.competitors.find(c => c.awayAway === 'away'); const hName = homeTeam.team.displayName; const aName = awayTeam.team.displayName; const hScore = parseInt(homeTeam.score) || 0; const aScore = parseInt(awayTeam.score) || 0; const statusType = event.status.type.state; // 'pre', 'in', 'post' // Versuche, aktuelle Torschützen-Details aus den Spiel-Details zu filtern, falls von ESPN geliefert let latestScorerText = ""; if (comp.details && comp.details.length > 0) { // Nimm das letzte Tor-Ereignis falls vorhanden const goalEvents = comp.details.filter(d => d.type && d.type.text && d.type.text.toLowerCase().includes('goal')); if (goalEvents.length > 0) { const lastGoal = goalEvents[goalEvents.length - 1]; latestScorerText = lastGoal.text ? ` (${lastGoal.text})` : ""; } } // Suche passendes Spiel in unseren Runden for (let rId in pokalMatchesByRound) { pokalMatchesByRound[rId].forEach(m => { if (m.home.toLowerCase().includes(hName.toLowerCase().substring(0, 5)) || hName.toLowerCase().includes(m.home.toLowerCase().substring(0, 5))) { let matchKey = `${m.home}-${m.away}`; let oldTotal = previousScores[matchKey] !== undefined ? previousScores[matchKey] : (m.homeScore || 0) + (m.awayScore || 0); let newTotal = hScore + aScore; m.homeScore = hScore; m.awayScore = aScore; if (statusType === 'post') m.isFinished = true; if (statusType === 'in') { hasLiveGames = true; m.isLive = true; } // Wenn ein Tor gefallen ist, detaillierten Push inklusive Torschützen-Info senden! if (statusType === 'in' && newTotal > oldTotal) { let pushBody = `${m.home} vs. ${m.away} — Neuer Spielstand: ${hScore}:${aScore}${latestScorerText}`; sendPushNotification("⚽ DFB-Pokal TOR!", pushBody); } previousScores[matchKey] = newTotal; } }); } }); } const liveBadge = document.getElementById('liveIndicatorBadge'); if (hasLiveGames) { liveBadge.style.display = 'flex'; } else { liveBadge.style.display = 'none'; } } catch (e) { console.log("Live-Daten Abruf im Offline-Modus / Fallback aktiv"); } renderMatches(); }
